def step1_hard_filter(match: MatchData) -> FilterResult:
    score = 0
    warnings = []
    
    # 赔率检查
    if match.odds.home_win < 1.30:
        return FilterResult(pass=False, reason="赔率过低(<1.30)")
    elif match.odds.home_win < 1.35:
        score -= 3
        warnings.append("赔率偏低")
    elif match.odds.home_win > 1.60:
        score -= 2
        warnings.append("赔率偏高")
    
    # 赛程检查
    if match.schedule.home_last_match <= "2天前" or match.schedule.away_last_match <= "2天前":
        return FilterResult(pass=False, reason="72小时内双赛")
    elif match.schedule.home_last_match <= "4天前" or match.schedule.away_last_match <= "4天前":
        score -= 2
        warnings.append("赛程密集")
    
    # 环境检查
    if match.environment.venue == "中立场" or match.environment.weather == "雨":
        return FilterResult(pass=False, reason="中立场/恶劣天气")
    
    # 战意检查
    if match.motivation.home == "弱" or match.motivation.away == "强":
        score -= 3
        warnings.append("战意存疑")
    
    return FilterResult(
        pass=score >= -2,
        score=score,
        warnings=warnings
    )

def step2_data_verify(match: MatchData) -> VerifyResult:
    # 进攻效率（40%）
    if match.stats.home_conversion_rate > 0.18:
        score1 = 100
    elif match.stats.home_conversion_rate > 0.15:
        score1 = 80
    elif match.stats.home_conversion_rate > 0.12:
        score1 = 60
    else:
        score1 = 30
    
    # 主场优势（35%）
    home_adv = match.stats.home_win_rate_home - match.stats.away_win_rate_away
    if home_adv > 0.30:
        score2 = 100
    elif home_adv > 0.20:
        score2 = 80
    elif home_adv > 0.10:
        score2 = 60
    else:
        score2 = 30
    
    # 交锋历史（25%）
    avg_corners = sum(h["corners"] for h in match.stats.head_to_head) / len(match.stats.head_to_head)
    avg_fouls = sum(h["fouls"] for h in match.stats.head_to_head) / len(match.stats.head_to_head)
    
    if avg_corners > 10 and avg_fouls < 15:
        score3 = 100
    elif avg_corners > 8 and avg_fouls < 20:
        score3 = 70
    else:
        score3 = 40
    
    total_score = score1 * 0.4 + score2 * 0.35 + score3 * 0.25
    
    if total_score >= 75:
        grade = "S"
    elif total_score >= 60:
        grade = "A"
    elif total_score >= 45:
        grade = "B"
    else:
        grade = "C"
    
    return VerifyResult(
        pass=total_score >= 60,
        score=total_score,
        grade=grade,
        details={
            "efficiency": score1,
            "home_adv": score2,
            "history": score3
        }
    )

def step3_market_verify(match: MatchData) -> MarketResult:
    signals = 0
    
    # 早期信号（赔率趋势）
    early_odds = match.market.odds_history[0].home_win
    current_odds = match.market.odds_history[-1].home_win
    odds_trend = early_odds - current_odds
    if odds_trend > 0.05:
        signals += 1  # 正向
    elif odds_trend < -0.05:
        signals -= 1  # 负向
    
    # 中期信号（欧亚同步）
    # 简化判断：赔率降，盘口退 = 背离
    if match.market.asian_handicap.initial == "让一球" and match.market.asian_handicap.current == "让半一":
        if odds_trend < 0:  # 赔率升
            signals -= 1
    
    # 后期信号（凯利趋势）
    early_kelly = match.market.kelly_history[0].home_win
    current_kelly = match.market.kelly_history[-1].home_win
    kelly_trend = early_kelly - current_kelly
    if kelly_trend > 0.03:
        signals += 1
    elif kelly_trend < -0.03:
        signals -= 1
    
    # 资金流向
    if match.market.money_flow == "正向":
        signals += 1
    elif match.market.money_flow == "反向":
        signals -= 1
    
    return MarketResult(
        pass=signals >= 0,
        score=signals,
        details={
            "odds_trend": odds_trend,
            "kelly_trend": kelly_trend,
            "money_flow": match.market.money_flow
        }
    )

def portfolio_optimizer(qualified_matches: List[QualifiedMatch], config: SystemConfig) -> Portfolio:
    # 按等级分组
    s_matches = [m for m in qualified_matches if m.grade == "S"]
    a_matches = [m for m in qualified_matches if m.grade == "A"]
    b_matches = [m for m in qualified_matches if m.grade == "B"]
    
    available_capital = config.total_capital * (1 - config.cash_reserve)
    
    portfolio = Portfolio()
    
    # 策略1：S级单选
    if s_matches:
        bet_size = available_capital * 0.30
        portfolio.add(
            type="single",
            match=s_matches[0],
            amount=bet_size
        )
        available_capital -= bet_size
    
    # 策略2：S+A 2串1
    if s_matches and a_matches:
        bet_size = available_capital * 0.40
        portfolio.add(
            type="2串1",
            matches=[s_matches[0], a_matches[0]],
            amount=bet_size
        )
        available_capital -= bet_size
    
    # 策略3：娱乐单
    if a_matches and b_matches:
        bet_size = available_capital * 0.20
        portfolio.add(
            type="2串1",
            matches=[a_matches[0], b_matches[0]],
            amount=bet_size
        )
    
    return portfolio

def safety_check(portfolio: Portfolio, history: List[BetHistory], config: SystemConfig) -> SafetyResult:
    warnings = []
    block = False
    
    # 检查1：单笔不超限
    for bet in portfolio.bets:
        if bet.amount > config.total_capital * config.max_single_bet:
            warnings.append(f"单笔超限：{bet.amount}")
            block = True
    
    # 检查2：每日不超限
    today_bets = sum(h.amount for h in history if h.date == today)
    if today_bets + portfolio.total > config.total_capital * config.daily_limit:
        warnings.append("今日限额已用完")
        block = True
    
    # 检查3：连黑熔断
    recent_losses = sum(1 for h in history[-3:] if h.result == "loss")
    if recent_losses >= config.consecutive_loss_stop:
        warnings.append("连黑熔断触发")
        block = True
    
    return SafetyResult(
        pass=not block,
        warnings=warnings,
        need_manual_confirm=True  # 永远需要人工确认
    )

def generate_daily_report(date: str, history: List[BetHistory]) -> Report:
    day_history = [h for h in history if h.date == date]
    
    stats = {
        "total_bets": len(day_history),
        "wins": sum(1 for h in day_history if h.result == "win"),
        "losses": sum(1 for h in day_history if h.result == "loss"),
        "roi": calculate_roi(day_history),
        "filtered_count": count_filtered_matches(date)
    }
    
    mistakes = []
    for h in day_history:
        if h.result == "loss":
            mistakes.append(analyze_mistake(h))
    
    return Report(
        date=date,
        stats=stats,
        mistakes=mistakes,
        suggestions=generate_suggestions(mistakes)
